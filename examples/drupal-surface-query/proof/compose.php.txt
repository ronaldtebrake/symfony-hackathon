$matches = [];
$counts = [];
foreach (['orchard', 'garden'] as $topic) {
    $result = $tools->drupal->search(['values' => ['keywords' => 'SCMQ '.$topic, 'limit' => 10]]);
    $counts[$topic] = $result->count;
    foreach ($result->matches as $match) {
        $matches[$match->id] = $match;
    }
}
ksort($matches);
return ['matches' => array_values($matches), 'count' => count($matches), 'returnedBySearch' => $counts];
